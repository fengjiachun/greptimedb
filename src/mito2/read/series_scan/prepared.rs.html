87        &self,
88        context: Arc<StreamContext>,
89        ranges: Vec<PartitionRange>,
90        metrics: PartitionMetrics,
91        scan_pool: Arc<dyn MemoryPool>,
92    ) -> Result<Arc<ColumnarPreparedSource>> {
93        let result = self
94            .source
95            .get_or_init(|| async {
96                let raw = RawScanPreparation {
97                    schema: Arc::new(datatypes::series::samples::SeriesSchema {
98                        fields: Vec::new().into(),
99                        eval_grid: None,
100                    }),
101                    fields: Vec::new(),
102                    predicates: Vec::new(),

104                ColumnarPreparedSource::prepare(
105                    context,
106                    ranges,
107                    RawLabelSource::Explicit,
108                    raw,
109                    metrics,
110                    scan_pool,
111                )
112                .await
113                .map(Arc::new)
114                .map_err(Arc::new)
115            })
116            .await;
117        let source = result
118            .as_ref()
119            .map_err(|error| crate::error::ScanSeriesSnafu.into_error(error.clone()))?;
120        if !matches!(source.labels, RawLabelSource::Explicit) {
121            return Err(invalid("cannot change prepared output representation"));
122        }
123        Ok(source.clone())
124    }

126    pub(crate) async fn prepare(
127        &self,
128        context: Arc<StreamContext>,
129        ranges: Vec<PartitionRange>,
130        labels: RawLabelSource,
131        raw: RawScanPreparation,
132        metrics: PartitionMetrics,
133        scan_pool: Arc<dyn MemoryPool>,
134    ) -> std::result::Result<Arc<dyn PreparedSeriesSource>, BoxedError> {
135        // Per caller, including time awaiting another partition's shared preparation.
136        // This overlaps prepare_wall and must not be added to its component timings.
137        let acquire_timer = crate::metrics::READ_STAGE_ELAPSED
138            .with_label_values(&["columnar_prepare_acquire_wall"])
139            .start_timer();
140        let expected = labels.clone();
141        let result = self
142            .source
143            .get_or_init(|| async {
144                ColumnarPreparedSource::prepare(context, ranges, labels, raw, metrics, scan_pool)
145                    .await
146                    .map(Arc::new)
147                    .map_err(Arc::new)
148            })
149            .await;
150        drop(acquire_timer);
151        let source = result.as_ref().map_err(|error| {
152            BoxedError::new(crate::error::ScanSeriesSnafu.into_error(error.clone()))
153        })?;
154        if !same_labels(&source.labels, &expected) {
155            return Err(BoxedError::new(invalid(
156                "label binding changed after series preparation",
157            )));
158        }
159        Ok(source.clone())
160    }

163pub(super) fn same_labels(left: &RawLabelSource, right: &RawLabelSource) -> bool {
164    match (left, right) {
165        (
166            RawLabelSource::PrimaryKey {
167                metric_name: a,
168                physical_metadata: am,
169                logical_tags: at,
170            },
171            RawLabelSource::PrimaryKey {
172                metric_name: b,
173                physical_metadata: bm,
174                logical_tags: bt,
175            },
176        ) => {
177            a == b
178                && Arc::ptr_eq(am, bm)
179                && at.len() == bt.len()
180                && at
181                    .iter()
182                    .zip(bt.iter())
183                    .all(|(a, b)| a.column_id == b.column_id && a.name == b.name)
184        }
185        _ => false,
186    }
187}

189/// Validates the pinned assignments and coverage without opening artifacts.
190///
191/// Returns whether every assigned source can be read from columnar artifacts. `false` is not an
192/// error: the caller scans those sources with the ordinary reader instead.
193pub(crate) fn validate_inputs(context: &StreamContext, ranges: &[PartitionRange]) -> Result<bool> {
194    if context.input.requires_narthex_preparation() {
195        narthex::file_indices(context, ranges)?;
196        return Ok(true);
197    }
198    collect_inputs(context, ranges).map(|inputs| inputs.is_some())
199}

201/// Collects the assigned SSTs and their covering columnar series indexes.
202///
203/// `None` means the assignments are structurally valid but not ready for columnar reads: an
204/// assigned memtable, an SST without columnar series coverage, or an SST without columnar
205/// samples. Structural corruption is still an error, and every assignment is validated before
206/// either answer is returned, so a later invalid assignment is never masked by missing artifacts.
207fn collect_inputs(
208    context: &StreamContext,
209    ranges: &[PartitionRange],
210) -> Result<Option<(Vec<usize>, HashSet<FileId>)>> {
211    let input = &context.input;
212    let metadata = input.region_metadata();
213    let index = input
214        .series_index
215        .as_ref()
216        .ok_or_else(|| invalid("missing columnar index context"))?;
217    let mut ready = true;
218    let mut files = BTreeMap::<usize, HashSet<i64>>::new();
219    for range in ranges {
220        let range = context
221            .ranges
222            .get(range.identifier)
223            .ok_or_else(|| invalid("invalid prepared scan range"))?;
224        for row_group in &range.row_group_indices {
225            if context.is_mem_range_index(*row_group) {
226                // Artifacts exist for flushed SSTs only.
227                ready = false;
228                continue;
229            }
230            if !context.is_file_range_index(*row_group) {
231                return Err(invalid(
232                    "columnar prepared reads do not support extension ranges",
233                ));
234            }
235            files
236                .entry(row_group.index - input.num_memtables())
237                .or_default()
238                .insert(row_group.row_group_index);
239        }
240    }
241    // Grouped artifacts may also contain sources excluded solely by time pruning.
242    // Sequence-pruned, unassigned, and retired sources remain ineligible.
243    let assigned: HashSet<_> = files
244        .keys()
245        .map(|i| input.files[*i].file_id().file_id())
246        .collect();
247    let allowed = assigned
248        .iter()
249        .chain(&input.time_pruned_files)
250        .copied()
251        .collect();
252    let eligible_samples = index
253        .version
254        .samples
255        .iter()
256        .filter(|(_, handle)| {
257            handle.entry().sources_visible(&allowed)
258                && assigned.iter().any(|id| handle.entry().covers(*id))
259        })
260        .map(|(id, handle)| (*id, handle.clone()))
261        .collect();
262    let mut indexes = HashSet::new();
263    let file_indices = files.keys().copied().collect();
264    for (file_idx, groups) in files {
265        let file = &input.files[file_idx];
266        if !groups.contains(&-1)
267            && (groups.len() as u64 != file.meta_ref().num_row_groups
268                || groups
269                    .iter()
270                    .any(|group| *group < 0 || *group as u64 >= file.meta_ref().num_row_groups))
271        {
272            return Err(invalid(
273                "columnar samples require complete SST range assignments",
274            ));
275        }
276        if file.meta_ref().region_id != metadata.region_id {
277            return Err(invalid(
278                "columnar prepared reads do not support foreign SSTs",
279            ));
280        }
281        let mut covered = false;
282        for handle in index.version.series_indexes.values() {
283            if handle.entry().format() == ArtifactFormat::ColumnarSeries
284                && handle
285                    .entry()
286                    .covers_file(file.meta_ref(), metadata.region_id)
287            {
288                indexes.insert(handle.entry().index_uuid);
289                covered = true;
290            }
291        }
292        if !covered {
293            ready = false;
294        }
295        if !crate::series_index::artifact::samples_cover_file(&eligible_samples, file)
296            || index.version.samples.values().any(|h| {
297                h.entry().covers(file.file_id().file_id())
298                    && h.entry().artifact.format != ArtifactFormat::ColumnarSamples
299            })
300        {
301            ready = false;
302        }
303    }
304    Ok(ready.then_some((file_indices, indexes)))
305}

307/// Pushes the supported conjuncts into the request.
308///
309/// The second element reports, for every input conjunct, whether `ColumnarReader`
310/// applies it exactly, so callers must not evaluate those conjuncts again.
311fn series_read_request(schema: &Schema, conjuncts: &[&Expr]) -> (SeriesReadRequest, Vec<bool>) {
312    let mut request = SeriesReadRequest {
313        projection: schema
314            .fields()
315            .iter()
316            .filter(|field| field.name() != ROW_COUNT_COLUMN)
317            .map(|field| field.name().clone())
318            .collect(),
319        ..Default::default()
320    };
321    let mut accepted = Vec::with_capacity(conjuncts.len());
322    for expr in conjuncts {
323        let Some(filter) = SimpleFilterEvaluator::try_new_with_column_type(expr, &|name| {
324            schema
325                .field_with_name(name)
326                .ok()
327                .map(|field| field.data_type().clone())
328        }) else {
329            accepted.push(false);
330            continue;
331        };
332        if !request
333            .projection
334            .iter()
335            .any(|name| name == filter.column_name())
336        {
337            accepted.push(false);
338            continue;
339        }
340        if filter.column_name() == DATA_SCHEMA_TABLE_ID_COLUMN_NAME
341            && filter.is_eq()
342            && let Some(Value::UInt32(table_id)) = filter.literal_value()
343        {
344            // Keep all equality filters, including contradictory ones, for exact evaluation.
345            request.table_id = Some(table_id);
346        }
347        request.filters.push(filter);
348        accepted.push(true);
349    }
350    (request, accepted)
351}

353/// Flattens the scanner predicate into the conjuncts shared by pushdown and residual filters.
354fn predicate_conjuncts(predicate: Option<&Predicate>) -> Vec<&Expr> {
355    predicate
356        .into_iter()
357        .flat_map(|predicate| predicate.exprs())
358        .flat_map(split_conjunction)
359        .collect()
360}

362struct Candidate {
363    /// Coarse prefix of the exact key; only a hash, never identity.
364    id: MetricSeriesId,
365    /// The one owned key allocation: the region-qualified exact key.
366    key: Buffer,
367    /// The canonical sparse key, a view of `key`'s bytes. This is the identity every
368    /// comparison and sample attribution uses, so it is stored once rather than once per
369    /// representation.
370    merge_key: Buffer,
371    tags: HashMap<u32, Option<String>>,
372    metric: Option<SeriesLabels>,
373}

391/// Opens sidecar readers after validating their complete source coverage.
392async fn open_sidecar_sources(
393    context: &StreamContext,
394    ranges: &[PartitionRange],
395) -> Result<(
396    Vec<Arc<dyn SamplesFormatReader>>,
397    Vec<Box<dyn FormatReader>>,
398)> {
399    let input = &context.input;
400    let metadata = input.region_metadata();
401    let index = input
402        .series_index
403        .as_ref()
404        .ok_or_else(|| invalid("missing columnar index context"))?;
405    let Some((files, indexes)) = collect_inputs(context, ranges)? else {
406        // Readiness was checked before preparation, so this is a real capability change.
407        return Err(invalid(
408            "columnar series coverage changed after series preparation",
409        ));
410    };
411    let cache = input
412        .cache_strategy
413        .columnar_metadata(index.cache_namespace);
414    let mut samples = Vec::<Arc<dyn SamplesFormatReader>>::new();
415    let assigned: HashSet<_> = files
416        .iter()
417        .map(|i| input.files[*i].file_id().file_id())
418        .collect();
419    let allowed = assigned
420        .iter()
421        .chain(&input.time_pruned_files)
422        .copied()
423        .collect();
424    for handle in index.version.samples.values().filter(|h| {
425        if !h.entry().sources_visible(&allowed) || !assigned.iter().any(|id| h.entry().covers(*id))
426        {
427            return false;
428        }
429        // Coverage validation above still needs every day of each source. Only
430        // opening readers can omit daily artifacts outside the query range.
431        if let (Some(day), Some(range)) = (h.entry().window, input.time_range.as_ref()) {
432            let bounds = day
433                .checked_mul(86_400)
434                .zip(day.checked_add(1).and_then(|next| next.checked_mul(86_400)));
435            if let Some((start, end)) = bounds {
436                return range.intersects(
437                    &(Timestamp::new_second(start)..Timestamp::new_second(end)).into(),
438                );
439            }
440            // Retain extreme days whose bounds cannot be represented in seconds.
441        }
442        true
443    }) {
444        let reader = crate::series_index::samples::reader::SamplesReader::open_cached(
445            index.store.clone(),
446            handle.clone(),
447            cache.clone(),
448        )
449        .await?;
450        format::validate_sample_schema(&reader.schema(), metadata)?;
451        samples.push(Arc::new(reader));
452    }
453    let mut series = Vec::<Box<dyn FormatReader>>::new();
454    for id in indexes {
455        series.push(Box::new(
456            crate::series_index::columnar::reader::ColumnarReader::open_cached(
457                index.store.clone(),
458                index.version.series_indexes[&id].clone(),
459                cache.clone(),
460            )
461            .await?,
462        ));
463    }
464    Ok((samples, series))
465}

467impl ColumnarPreparedSource {
468    async fn prepare(
469        context: Arc<StreamContext>,
470        ranges: Vec<PartitionRange>,
471        labels: RawLabelSource,
472        raw: RawScanPreparation,
473        metrics: PartitionMetrics,
474        scan_pool: Arc<dyn MemoryPool>,
475    ) -> Result<Self> {
476        let _prepare_timer = crate::metrics::READ_STAGE_ELAPSED
477            .with_label_values(&["columnar_prepare_wall"])
478            .start_timer();
479        let mut preparation_metrics = crate::metrics::ColumnarReadMetrics::default();
480        preparation_metrics.counts.insert("prepare_executions", 1);
481        let started = Instant::now();
482        let input = &context.input;
483        let metadata = input.region_metadata();
484        let (samples, series) = if context.input.requires_narthex_preparation() {
485            narthex::open_sources(&context, &ranges, &scan_pool).await?
486        } else {
487            open_sidecar_sources(&context, &ranges).await?
488        };
489        preparation_metrics.add_elapsed("columnar_prepare_coverage", started.elapsed());
490        preparation_metrics
491            .counts
492            .insert("prepare_sample_readers", samples.len() as u64);
493        preparation_metrics
494            .counts
495            .insert("prepare_series_indexes", series.len() as u64);
496        let reservation = MemoryConsumer::new("ColumnarPreparedSource").register(&scan_pool);
497        let started = Instant::now();
498        let mut raw = raw;
499        let batch_output = matches!(labels, RawLabelSource::Explicit);
500        let dedup_append_raw =
501            input.requires_narthex_preparation() && input.append_mode && !batch_output;
502        let sample_schema = if batch_output {
503            Some(input.mapper.input_arrow_schema(input.compaction))
504        } else {
505            sample_merge_schema(&context, &mut raw)?
506        };
507        // Sparse compaction omits tag columns, unlike the mapper's query batch.
508        // Derive the offset from the schema that actually reaches the dedup reader.
509        let merge_field_start = sample_schema.as_ref().map_or_else(
510            || input.mapper.field_column_start(),
511            |schema| {
512                schema
513                    .fields()
514                    .iter()
515                    .position(|field| {
516                        metadata
517                            .column_by_name(field.name())
518                            .is_some_and(|column| column.semantic_type == SemanticType::Field)
519                    })
520                    .unwrap_or(0)
521            },
522        );
523        preparation_metrics.add_elapsed("columnar_prepare_schema", started.elapsed());
524        preparation_metrics.bytes.insert("prepare_encoded_keys", 0);
525        preparation_metrics.counts.insert("prepare_batch_series", 0);
526        let direct_labels = !batch_output && sample_schema.is_some();
527        let conjuncts = predicate_conjuncts(input.predicate_group().predicate());
528        let mut retained_rows = 0_usize;
529        let mut candidates = BTreeMap::<ExactSeriesKey, Candidate>::new();
530        for reader in series {
531            let started = Instant::now();
532            let schema = reader.schema();
533            format::validate_series_schema(&schema, metadata)?;
534            let mut tags = schema
535                .fields()
536                .iter()
537                .skip(3)
538                .map(|field| {
539                    let id = format::field_id(field)?;
540                    Ok((id, field.name().clone()))
541                })
542                .collect::<Result<Vec<_>>>()?;
543            tags.sort_unstable_by(|a, b| a.1.cmp(&b.1));
544            let (request, accepted) = series_read_request(&schema, &conjuncts);
545            let projection = request
546                .projection
547                .iter()
548                .map(|name| schema.index_of(name).map_err(|e| invalid(e.to_string())))
549                .collect::<Result<Vec<_>>>()?;
550            let projected_schema = Arc::new(
551                schema
552                    .project(&projection)
553                    .map_err(|e| invalid(e.to_string()))?,
554            );
555            let mut filters = Vec::new();
556            for (expr, accepted) in conjuncts.iter().zip(&accepted) {
557                // The reader already applied accepted conjuncts exactly.
558                if *accepted
559                    || !expr
560                        .column_refs()
561                        .iter()
562                        .all(|column| projected_schema.index_of(&column.name).is_ok())
563                {
564                    continue;
565                }
566                filters.push(
567                    Predicate::to_physical_expr(expr, &projected_schema)
568                        .map_err(|e| invalid(e.to_string()))?,
569                );
570            }
571            if context.input.requires_narthex_preparation()
572                && let Some(partition) =
573                    narthex_partition_filter(metadata.partition_expr.as_deref(), None)?
574            {
575                filters.push(
576                    partition
577                        .try_as_physical_expr(&projected_schema)
578                        .map_err(|error| invalid(error.to_string()))?,
579                );
580            }
581            preparation_metrics.add_elapsed("columnar_prepare_compile", started.elapsed());
582            let mut batches = reader.read(request)?;
583            loop {
584                let timer = preparation_metrics.timer("columnar_prepare_discovery_wall");
585                let batch = batches.try_next().await?;
586                drop(timer);
587                let Some(mut batch) = batch else {
588                    break;
589                };
590                *preparation_metrics
591                    .counts
592                    .entry("prepare_discovered_rows")
593                    .or_default() += batch.num_rows() as u64;
594                *preparation_metrics
595                    .counts
596                    .entry("prepare_discovery_batches")
597                    .or_default() += 1;
598                let started = Instant::now();
599                for filter in &filters {
600                    batch = common_recordbatch::filter::batch_filter(&batch, filter)
601                        .map_err(|e| invalid(e.to_string()))?;
602                }
603                preparation_metrics.add_elapsed("columnar_prepare_residual", started.elapsed());
604                *preparation_metrics
605                    .counts
606                    .entry("prepare_selected_rows")
607                    .or_default() += batch.num_rows() as u64;
608                let started = Instant::now();
609                let mut duplicate_rows = 0_u64;
610                let table = batch
611                    .column_by_name(DATA_SCHEMA_TABLE_ID_COLUMN_NAME)
612                    .ok_or_else(|| invalid("missing table ID column"))?
613                    .as_any()
614                    .downcast_ref::<UInt32Array>()
615                    .ok_or_else(|| invalid("invalid table ID column"))?;
616                let tsid = batch
617                    .column_by_name(DATA_SCHEMA_TSID_COLUMN_NAME)
618                    .ok_or_else(|| invalid("missing TSID column"))?
619                    .as_any()
620                    .downcast_ref::<UInt64Array>()
621                    .ok_or_else(|| invalid("invalid TSID column"))?;
622                let tag_arrays = tags
623                    .iter()
624                    .map(|(id, name)| {
625                        let strings = batch
626                            .column_by_name(name)
627                            .ok_or_else(|| invalid("missing series tag column"))?
628                            .as_any()
629                            .downcast_ref::<StringArray>()
630                            .ok_or_else(|| invalid("invalid series tag array"))?;
631                        Ok((*id, strings))
632                    })
633                    .collect::<Result<Vec<_>>>()?;
634                // Identity is the exact canonical sparse key. `tsid` is a hash, so rows
635                // that differ only in labels are different series and must both survive:
636                // re-encoding the artifact's own tags reproduces the SST key byte for byte.
637                let codec = SparsePrimaryKeyCodec::schemaless();
638                let mut offsets = mito_codec::row_converter::SparseOffsetsCache::new();
639                let mut identity = Vec::new();
640                let mut previous: Option<Vec<u8>> = None;
641                let mut fresh = Vec::<(ExactSeriesKey, usize)>::new();
642                let mut encoded_key_bytes = 0_u64;
643                for row in 0..batch.num_rows() {
644                    identity.clear();
645                    codec
646                        .encode_internal(table.value(row), tsid.value(row), &mut identity)
647                        .map_err(|e| invalid(e.to_string()))?;
648                    codec
649                        .encode_to_vec(
650                            tag_arrays.iter().map(|(id, strings)| {
651                                (
652                                    *id,
653                                    if strings.is_null(row) {
654                                        ValueRef::Null
655                                    } else {
656                                        ValueRef::String(strings.value(row))
657                                    },
658                                )
659                            }),
660                            &mut identity,
661                        )
662                        .map_err(|e| invalid(e.to_string()))?;
663                    encoded_key_bytes += identity.len() as u64;
664                    // Adjacent duplicates are the common case and never allocate a key.
665                    if previous.as_deref() == Some(identity.as_slice())
666                        || candidates.contains_key(identity.as_slice())
667                    {
668                        duplicate_rows += 1;
669                        continue;
670                    }
671                    let key = ExactSeriesKey::from_sparse_key(&identity, &mut offsets)?;
672                    previous = Some(identity.clone());
673                    fresh.push((key, row));
674                }
675                // Retain only rows that introduce a new identity; the retained batch backs
676                // both explicit labels and the metadata rows of direct-label candidates.
677                let retained = if fresh.len() == batch.num_rows() {
678                    batch.clone()
679                } else {
680                    let selected: Vec<u64> = fresh.iter().map(|(_, row)| *row as u64).collect();
681                    take_record_batch(&batch, &UInt64Array::from(selected))
682                        .context(ComputeArrowSnafu)?
683                };
684                if direct_labels {
685                    // Only explicit-label chunks retain a batch to resolve labels from.
686                    retained_rows += retained.num_rows();
687                }
688                let label_batch = if direct_labels {
689                    let RawLabelSource::PrimaryKey { logical_tags, .. } = &labels else {
690                        return Err(invalid("expected bound labels"));
691                    };
692                    let columns = logical_tags
693                        .iter()
694                        .map(|tag| {
695                            let name = tags
696                                .iter()
697                                .find(|(id, _)| *id == tag.column_id)
698                                .ok_or_else(|| invalid("missing bound tag"))?;
699                            Ok((
700                                tag.name.clone(),
701                                retained
702                                    .schema()
703                                    .index_of(&name.1)
704                                    .map_err(|e| invalid(e.to_string()))?,
705                            ))
706                        })
707                        .collect::<Result<Vec<_>>>()?;
708                    let owner = Arc::new(
709                        MemoryConsumer::new("ColumnarSeriesLabelBatch").register(&scan_pool),
710                    );
711                    let label_batch =
712                        SeriesLabelBatch::try_new(retained.clone(), columns, owner.clone())
713                            .map_err(|e| invalid(e.to_string()))?;
714                    owner
715                        .try_grow(label_batch.estimated_size())
716                        .context(crate::error::ScanMemoryExhaustedSnafu)?;
717                    Some(Arc::new(label_batch))
718                } else {
719                    None
720                };
721                let mut tag_values = 0_u64;
722                let mut batch_series = 0_u64;
723                for (retained_row, (exact, row)) in fresh.into_iter().enumerate() {
724                    let mut values = HashMap::new();
725                    if !direct_labels && (!batch_output || !input.compaction) {
726                        values.reserve(tag_arrays.len());
727                        for (column_id, strings) in &tag_arrays {
728                            values.insert(
729                                *column_id,
730                                (!strings.is_null(row)).then(|| strings.value(row).to_owned()),
731                            );
732                        }
733                    }
734                    tag_values += values.len() as u64;
735                    // `merge_key` is always the exact key bytes. Explicit-label chunks keep
736                    // the region-qualified form as their `key`, so `primary_key` stays
737                    // region-scoped while identity and ordering stay exact.
738                    let key = if direct_labels {
739                        let mut buffer = Vec::with_capacity(8 + exact.bytes().len());
740                        buffer.extend_from_slice(&metadata.region_id.as_u64().to_be_bytes());
741                        buffer.extend_from_slice(exact.bytes());
742                        Buffer::from(buffer)
743                    } else {
744                        Buffer::from(exact.bytes().to_vec())
745                    };
746                    let merge_key = if direct_labels {
747                        key.slice_with_length(8, exact.bytes().len())
748                    } else {
749                        key.clone()
750                    };
751                    // The exact key is stored once: `merge_key` is a view of `key`, so the
752                    // reservation charges the single key allocation the candidate retains.
753                    let bytes = key.len()
754                        + values.values().flatten().map(String::len).sum::<usize>()
755                        + values.capacity() * (std::mem::size_of::<(u32, Option<String>)>() + 16)
756                        + std::mem::size_of::<Candidate>()
757                        + 64;
758                    reservation
759                        .try_grow(bytes)
760                        .context(crate::error::ScanMemoryExhaustedSnafu)?;
761                    let metric = if let Some(label_batch) = &label_batch {
762                        batch_series += 1;
763                        let RawLabelSource::PrimaryKey { metric_name, .. } = &labels else {
764                            return Err(invalid("expected bound labels"));
765                        };
766                        Some(
767                            SeriesLabels::from_batch(
768                                Some(metric_name.clone()),
769                                label_batch.clone(),
770                                retained_row,
771                            )
772                            .map_err(|e| invalid(e.to_string()))?,
773                        )
774                    } else {
775                        None
776                    };
777                    let id = exact.id();
778                    candidates.insert(
779                        exact,
780                        Candidate {
781                            id,
782                            key,
783                            merge_key,
784                            tags: values,
785                            metric,
786                        },
787                    );
788                }
789                *preparation_metrics
790                    .bytes
791                    .entry("prepare_encoded_keys")
792                    .or_default() += encoded_key_bytes;
793                *preparation_metrics
794                    .counts
795                    .entry("prepare_tag_values")
796                    .or_default() += tag_values;
797                *preparation_metrics
798                    .counts
799                    .entry("prepare_duplicate_rows")
800                    .or_default() += duplicate_rows;
801                *preparation_metrics
802                    .counts
803                    .entry("prepare_batch_series")
804                    .or_default() += batch_series;
805                preparation_metrics.add_elapsed("columnar_prepare_candidates", started.elapsed());
806            }
807        }
808        let started = Instant::now();
809        preparation_metrics
810            .counts
811            .insert("prepare_unique_series", candidates.len() as u64);
812        preparation_metrics
813            .counts
814            .insert("prepare_retained_rows", retained_rows as u64);
815        let mut candidates: Vec<_> = candidates.into_values().collect();
816        candidates.sort_unstable_by(|a, b| a.key.as_slice().cmp(b.key.as_slice()));
817        preparation_metrics.add_elapsed("columnar_prepare_sort", started.elapsed());
818        Ok(Self {
819            context,
820            labels,
821            raw: Arc::new(raw),
822            sample_schema,
823            merge_field_start,
824            dedup_append_raw,
825            candidates: Arc::new(candidates),
826            samples: Arc::new(samples),
827            metrics,
828            scan_pool,
829            reservation: Arc::new(reservation),
830        })
831    }
832}

834impl ColumnarPreparedSource {
835    pub(crate) fn read_batches(&self, range: Range<usize>) -> Result<BoxedRecordBatchStream> {
836        if range.start > range.end || range.end > self.candidates.len() {
837            return Err(invalid("series range out of bounds"));
838        }
839        let context = self.context.clone();
840        let sample_schema = self.sample_schema.clone();
841        let candidates = self.candidates.clone();
842        let samples = self.samples.clone();
843        let metrics = self.metrics.clone();
844        let reservation = self.reservation.clone();
845        let scan_pool = self.scan_pool.clone();
846        let field_column_start = self.merge_field_start;
847        let dedup_append_raw = self.dedup_append_raw;
848        let filters =
849            if matches!(self.labels, RawLabelSource::Explicit) && !context.input.compaction {
850                predicate_conjuncts(context.input.predicate_group().predicate())
851                    .into_iter()
852                    .map(|expr| {
853                        Predicate::to_physical_expr(
854                            expr,
855                            &context.input.mapper.input_arrow_schema(false),
856                        )
857                        .map_err(|error| invalid(error.to_string()))
858                    })
859                    .collect::<Result<Vec<_>>>()?
860            } else {
861                Vec::new()
862            };
863        let stream = async_stream::try_stream! {
864            let _reservation = reservation;
865            if range.is_empty() { return; }
866            let mut sources = Vec::<BoxedRecordBatchStream>::new();
867            for reader in samples.iter() {
868                let candidates = candidates.clone();
869                let range = range.clone();
870                let context = context.clone();
871                let reader = reader.clone();
872                let sample_schema = sample_schema.clone();
873                let scan_pool = scan_pool.clone();
874                sources.push(Box::pin(async_stream::try_stream! {
875                    let reservation = MemoryConsumer::new("ColumnarSamplesRead").register(&scan_pool);
876                    // Account for the selection retained by the samples reader, including
877                    // B-tree node overhead, before allocating it. The exact-key selection and
878                    // the coarse identifiers the reader derives from it both live for the
879                    // lifetime of the sample stream.
880                    let selection_bytes = candidates[range.clone()]
881                        .iter()
882                        .map(|candidate| {
883                            std::mem::size_of::<ExactSeriesKey>()
884                                + candidate.merge_key.len()
885                                + std::mem::size_of::<MetricSeriesId>()
886                                + 64
887                        })
888                        .sum::<usize>();
889                    reservation.try_grow(selection_bytes).context(crate::error::ScanMemoryExhaustedSnafu)?;
890                    // Exact ownership: a block is read only when its own key is selected,
891                    // never because a colliding partner shares the identifier. Coarse
892                    // pruning bounds are derived from these keys inside the reader.
893                    // The keys are built from the one key the candidate retains, so the
894                    // selection costs one owned copy per series rather than a second copy
895                    // alongside every candidate.
896                    let mut offsets = mito_codec::row_converter::SparseOffsetsCache::new();
897                    let keys = candidates[range.clone()]
898                        .iter()
899                        .map(|c| ExactSeriesKey::from_sparse_key(c.merge_key.as_slice(), &mut offsets))
900                        .collect::<Result<BTreeSet<_>>>()?;
901                    let mut stage_metrics = crate::metrics::ColumnarReadMetrics::default();
902                    let schema = reader.schema();
903                    let metadata = context.input.region_metadata();
904                    let name = &metadata.time_index_column().column_schema.name;
905                    let field = schema.field_with_name(name).map_err(|err| invalid(err.to_string()))?;
906                    let unit = match field.data_type() {
907                        datatypes::arrow::datatypes::DataType::Timestamp(unit, _) => Ok(common_time::timestamp::TimeUnit::from(unit)),
908                        _ => Err(invalid(format!("sample time index {name} must be a timestamp"))),
909                    }?;
910                    let bounds = context.input.time_range.as_ref();
911                    let start = bounds.and_then(|r| *r.start()).and_then(|t| t.convert_to_ceil(unit)).map(|t| t.value());
912                    let end = bounds.and_then(|r| *r.end()).and_then(|t| t.convert_to_ceil(unit)).map(|t| t.value());
913                    let mut batches = reader.read(SamplesReadRequest {
914                        projection: sample_projection(&schema, &context),
915                        memory_pool: Some(scan_pool.clone()),
916                        keys: Some(keys),
917                        start,
918                        end,
919                        ..Default::default()
920                    })?;
921                    while let Some(samples) = batches.try_next().await? {
922                        // A block belongs to exactly one series, so the reader returns its
923                        // exact key once and the whole batch is attributed at once.
924                        let SamplesBatch { batch, key } = samples;
925                        let ordinal = candidates[range.clone()]
926                            .binary_search_by(|candidate| {
927                                candidate.merge_key.as_slice().cmp(key.bytes())
928                            })
929                            .map_err(|_| invalid("sample contains an unselected series"))?
930                            + range.start;
931                        let started = std::time::Instant::now();
932                        let flat = sample_to_flat(batch, &candidates[ordinal], &context, sample_schema.as_ref())?;
933                        let flat = filter_visibility(flat, &context)?;
934                        stage_metrics.add_elapsed("columnar_sample_adapt", started.elapsed());
935                        if flat.num_rows() == 0 { continue; }
936                        let bytes = flat.get_array_memory_size();
937                        reservation.try_grow(bytes).context(crate::error::ScanMemoryExhaustedSnafu)?;
938                        yield flat;
939                        reservation.shrink(bytes);
940                    }
941                }));
942            }
943            let mut merged = if let Some(schema) = sample_schema {
944                SeqScan::merge_flat_sources(&context, sources, Some(&metrics), false, schema, field_column_start).await?
945            } else {
946                SeqScan::build_flat_reader_from_sources(&context, sources, None, Some(&metrics), false, 1).await?
947            };
948            // Append mode skips dedup in the merge reader, but raw series reads
949            // require unique timestamps. Non-append reads use the configured merge
950            // mode above; batch reads retain append semantics.
951            if dedup_append_raw {
952                merged = Box::pin(crate::read::flat_dedup::FlatDedupReader::new(
953                    merged,
954                    crate::read::flat_dedup::FlatLastRow::new(context.input.filter_deleted),
955                    Some(metrics.dedup_metrics_reporter()),
956                ).into_stream());
957            }
958            while let Some(mut batch) = merged.try_next().await? {
959                for filter in &filters {
960                    batch = common_recordbatch::filter::batch_filter(&batch, filter)
961                        .map_err(|error| invalid(error.to_string()))?;
962                }
963                if batch.num_rows() > 0 {
964                    yield batch;
965                }
966            }
967        };
968        Ok(Box::pin(stream))
969    }
970}

972impl PreparedSeriesSource for ColumnarPreparedSource {
973    fn num_series(&self) -> usize {
974        self.candidates.len()
975    }
976    fn label_source(&self) -> &RawLabelSource {
977        &self.labels
978    }
979    fn series_info(&self, ordinal: usize) -> std::result::Result<SeriesInfo, BoxedError> {
980        let candidate = self
981            .candidates
982            .get(ordinal)
983            .ok_or_else(|| BoxedError::new(invalid("series ordinal out of bounds")))?;
984        Ok(SeriesInfo {
985            primary_key: candidate.key.clone(),
986            metric: candidate.metric.clone(),
987        })
988    }
989    fn read_series(
990        &self,
991        range: Range<usize>,
992    ) -> std::result::Result<SendableRawSeriesStream, BoxedError> {
993        let mut batches = self.read_batches(range).map_err(BoxedError::new)?;
994        let stream = async_stream::try_stream! {
995            while let Some(batch) = batches.try_next().await? {
996                yield ScanBatch::RecordBatch(batch);
997            }
998        };
999        let stream = raw::make_stream(
1000            Box::pin(stream),
1001            self.raw.schema.clone(),
1002            self.labels.clone(),
1003            self.raw.fields.clone(),
1004            self.metrics.clone(),
1005            self.raw.predicates.clone(),
1006            SeriesOrdering::KeyThenTime,
1007        );
1008        if self.sample_schema.is_none() {
1009            return Ok(stream);
1010        }
1011        let candidates = self.candidates.clone();
1012        let reservation = self.reservation.clone();
1013        Ok(raw::with_series_labels(stream, move |key| {
1014            // The resolver can outlive the exhausted sample stream.
1015            let _reservation = &reservation;
1016            let ordinal = candidates
1017                .binary_search_by(|candidate| candidate.merge_key.as_slice().cmp(key.as_slice()))
1018                .map_err(|_| BoxedError::new(invalid("unknown merged series identity")))?;
1019            Ok((
1020                candidates[ordinal].key.clone(),
1021                candidates[ordinal]
1022                    .metric
1023                    .clone()
1024                    .ok_or_else(|| BoxedError::new(invalid("identity merge is missing labels")))?,
1025            ))
1026        }))
1027    }
1028}

1030/// Omits tags only when every predicate is enforced either on the series index
1031/// or on sample columns. Conjunctions can be separated; mixed disjunctions
1032/// retain the ordinary flat adapter.
1033pub(crate) fn sample_merge_schema(
1034    context: &StreamContext,
1035    raw: &mut RawScanPreparation,
1036) -> Result<Option<SchemaRef>> {
1037    let metadata = context.input.region_metadata();
1038    let full = context.input.mapper.input_arrow_schema(false);
1039    let projection: Vec<_> = full
1040        .fields()
1041        .iter()
1042        .enumerate()
1043        .filter_map(|(i, field)| {
1044            let tag = metadata
1045                .column_by_name(field.name())
1046                .is_some_and(|column| column.semantic_type == SemanticType::Tag);
1047            (!tag).then_some(i)
1048        })
1049        .collect();
1050    let schema = Arc::new(full.project(&projection).context(ComputeArrowSnafu)?);
1051    let mut predicates = Vec::new();
1052    if let Some(predicate) = context.input.predicate_group().predicate() {
1053        for expr in predicate.exprs().iter().flat_map(split_conjunction) {
1054            let columns = expr.column_refs();
1055            if columns.iter().all(|column| {
1056                metadata
1057                    .column_by_name(&column.name)
1058                    .is_some_and(|column| column.semantic_type == SemanticType::Tag)
1059            }) {
1060                // prepare() applies this complete expression to every series-index batch.
1061                continue;
1062            }
1063            if !columns
1064                .iter()
1065                .all(|column| schema.index_of(&column.name).is_ok())
1066            {
1067                return Ok(None);
1068            }
1069            predicates.push(
1070                Predicate::to_physical_expr(expr, &schema).map_err(|e| invalid(e.to_string()))?,
1071            );
1072        }
1073    }
1074    let fields = raw
1075        .fields
1076        .iter()
1077        .map(|(index, data_type)| {
1078            projection
1079                .iter()
1080                .position(|old| old == index)
1081                .map(|index| (index, data_type.clone()))
1082                .ok_or_else(|| invalid("sample merge is missing a selected field"))
1083        })
1084        .collect::<Result<Vec<_>>>()?;
1085    raw.fields = fields;
1086    raw.predicates = predicates;
1087    Ok(Some(schema))
1088}

1090fn sample_to_flat(
1091    batch: RecordBatch,
1092    candidate: &Candidate,
1093    context: &StreamContext,
1094    sample_schema: Option<&SchemaRef>,
1095) -> Result<RecordBatch> {
1096    let schema = sample_schema
1097        .cloned()
1098        .unwrap_or_else(|| context.input.mapper.input_arrow_schema(false));
1099    let key = if sample_schema.is_some() {
1100        &candidate.merge_key
1101    } else {
1102        &candidate.key
1103    };
1104    let metadata = context.input.region_metadata();
1105    let mut arrays = Vec::<ArrayRef>::new();
1106    for field in schema.fields() {
1107        let array: ArrayRef = match field.name().as_str() {
1108            "__primary_key" => Arc::new(
1109                DictionaryArray::<UInt32Type>::try_new(
1110                    UInt32Array::from(vec![0; batch.num_rows()]),
1111                    Arc::new(BinaryArray::new(
1112                        OffsetBuffer::new(
1113                            vec![
1114                                0,
1115                                i32::try_from(key.len()).map_err(|_| {
1116                                    invalid("series key exceeds Arrow offset limit")
1117                                })?,
1118                            ]
1119                            .into(),
1120                        ),
1121                        key.clone(),
1122                        None,
1123                    )),
1124                )
1125                .context(ComputeArrowSnafu)?,
1126            ),
1127            "__sequence" | "__op_type" => batch
1128                .column_by_name(field.name())
1129                .ok_or_else(|| invalid("missing sample internal column"))?
1130                .clone(),
1131            name => {
1132                let column = metadata
1133                    .column_by_name(name)
1134                    .ok_or_else(|| invalid("unknown flat column"))?;
1135                if column.semantic_type == SemanticType::Tag {
1136                    if column.column_id == ReservedColumnId::table_id() {
1137                        Arc::new(UInt32Array::from(vec![
1138                            candidate.id.table_id;
1139                            batch.num_rows()
1140                        ]))
1141                    } else if column.column_id == ReservedColumnId::tsid() {
1142                        Arc::new(UInt64Array::from(vec![candidate.id.tsid; batch.num_rows()]))
1143                    } else {
1144                        let value = candidate
1145                            .tags
1146                            .get(&column.column_id)
1147                            .ok_or_else(|| invalid("missing series tag"))?;
1148                        Arc::new(StringArray::from(vec![value.as_deref(); batch.num_rows()]))
1149                    }
1150                } else {
1151                    match batch.column_by_name(name) {
1152                        Some(array) => array.clone(),
1153                        None if field.is_nullable() => datatypes::arrow::array::new_null_array(
1154                            field.data_type(),
1155                            batch.num_rows(),
1156                        ),
1157                        None => return Err(invalid("missing non-null sample field")),
1158                    }
1159                }
1160            }
1161        };
1162        arrays.push(if array.data_type() == field.data_type() {
1163            array
1164        } else {
1165            cast(&array, field.data_type()).context(ComputeArrowSnafu)?
1166        });
1167    }
1168    RecordBatch::try_new(schema, arrays).context(NewRecordBatchSnafu)
1169}

1171fn filter_visibility(batch: RecordBatch, context: &StreamContext) -> Result<RecordBatch> {
1172    let sequence_column = batch
1173        .column_by_name(SEQUENCE_COLUMN_NAME)
1174        .ok_or_else(|| invalid("missing sequence column __sequence"))?;
1175    let sequence = sequence_column
1176        .as_any()
1177        .downcast_ref::<UInt64Array>()
1178        .ok_or_else(|| invalid("sequence column __sequence must be UInt64"))?;
1179    let metadata = context.input.region_metadata();
1180    let timestamp = &metadata.time_index_column().column_schema;
1181    let time_column = batch
1182        .column_by_name(&timestamp.name)
1183        .ok_or_else(|| invalid(format!("missing time index column {}", timestamp.name)))?;
1184    let unit = metadata.time_index_type().unit();
1185    let (time, actual_unit) = datatypes::timestamp::timestamp_array_to_primitive(time_column)
1186        .ok_or_else(|| {
1187            invalid(format!(
1188                "time index column {} must be a timestamp",
1189                timestamp.name
1190            ))
1191        })?;
1192    if common_time::timestamp::TimeUnit::from(actual_unit) != unit {
1193        return Err(invalid(format!(
1194            "time index column {} has unit {actual_unit:?}, expected {unit:?}",
1195            timestamp.name
1196        )));
1197    }
1198    // Bounds are computed from the actual arrays: sample metadata does not expose
1199    // sequence bounds here, and format-reader timestamps need not be sorted.
1200    if sequence.null_count() == 0 && time.null_count() == 0 {
1201        let needs_sequence_max = context.input.snapshot_sequence.is_some()
1202            || matches!(
1203                context.input.sequence_range,
1204                Some(SequenceRange::LtEq { .. } | SequenceRange::GtLtEq { .. })
1205            );
1206        let sequence_max = needs_sequence_max.then(|| max(sequence)).flatten();
1207        let snapshot_visible = context
1208            .input
1209            .snapshot_sequence
1210            .is_none_or(|bound| sequence_max.is_some_and(|value| value <= bound));
1211        let range_visible = context
1212            .input
1213            .sequence_range
1214            .is_none_or(|range| match range {
1215                SequenceRange::Gt { min: bound } => {
1216                    min(sequence).is_some_and(|value| value > bound)
1217                }
1218                SequenceRange::LtEq { max: bound } => {
1219                    sequence_max.is_some_and(|value| value <= bound)
1220                }
1221                SequenceRange::GtLtEq {
1222                    min: lower,
1223                    max: upper,
1224                } => {
1225                    sequence_max.is_some_and(|value| value <= upper)
1226                        && min(sequence).is_some_and(|value| value > lower)
1227                }
1228            });
1229        let time_visible = context.input.time_range.as_ref().is_none_or(|range| {
1230            min(&time).zip(max(&time)).is_some_and(|(lower, upper)| {
1231                range.contains(&common_time::Timestamp::new(lower, unit))
1232                    && range.contains(&common_time::Timestamp::new(upper, unit))
1233            })
1234        });
1235        if snapshot_visible && range_visible && time_visible {
1236            return Ok(batch);
1237        }
1238    }
1239    let sequence_filter = context
1240        .input
1241        .sequence_range
1242        .map(|range| range.filter(sequence_column))
1243        .transpose()
1244        .context(ComputeArrowSnafu)?;
1245    let filter = BooleanArray::from(
1246        (0..batch.num_rows())
1247            .map(|i| {
1248                !sequence.is_null(i)
1249                    && !time.is_null(i)
1250                    && context
1251                        .input
1252                        .snapshot_sequence
1253                        .is_none_or(|max| sequence.value(i) <= max)
1254                    && sequence_filter
1255                        .as_ref()
1256                        .is_none_or(|filter| filter.value(i))
1257                    && context.input.time_range.as_ref().is_none_or(|range| {
1258                        range.contains(&common_time::Timestamp::new(time.value(i), unit))
1259                    })
1260            })
1261            .collect::<Vec<_>>(),
1262    );
1263    filter_record_batch(&batch, &filter).context(ComputeArrowSnafu)
1264}

1266/// The mapper includes projected and residual-filter columns. Identity/time and
1267/// merge visibility columns are always needed by the narthex source adapter. The exact
1268/// series key is not projected: the samples reader returns it once per batch.
1269fn sample_projection(
1270    schema: &datatypes::arrow::datatypes::Schema,
1271    context: &StreamContext,
1272) -> Vec<String> {
1273    let input = context.input.mapper.input_arrow_schema(false);
1274    schema
1275        .fields()
1276        .iter()
1277        .enumerate()
1278        .filter(|(index, field)| {
1279            field.name() != store_api::storage::consts::PRIMARY_KEY_COLUMN_NAME
1280                && (*index < 5 || input.index_of(field.name()).is_ok())
1281        })
1282        .map(|(_, field)| field.name().clone())
1283        .collect()
1284}

1286/// Returns a partition rule only when metrics rows need filtering for this owner.
1287fn narthex_partition_filter(
1288    region_partition: Option<&str>,
1289    file_partition: Option<&partition::expr::PartitionExpr>,
1290) -> Result<Option<partition::expr::PartitionExpr>> {
1291    Ok(region_partition
1292        .map(|json| {
1293            partition::expr::PartitionExpr::from_json_str(json)
1294                .map_err(|error| invalid(error.to_string()))
1295        })
1296        .transpose()?
1297        .flatten()
1298        .filter(|partition| file_partition != Some(partition)))
1299}

1301/// Shares narthex discovery across partitions and assigns disjoint exact-series ranges.
1302pub(crate) fn scan_batches(
1303    preparation: Arc<Preparation>,
1304    context: Arc<StreamContext>,
1305    ranges: Vec<PartitionRange>,
1306    partition: usize,
1307    partitions: usize,
1308    metrics: PartitionMetrics,
1309    scan_pool: Arc<dyn MemoryPool>,
1310) -> crate::read::stream::ScanBatchStream {
1311    Box::pin(async_stream::try_stream! {
1312        let source = preparation.prepare_batches(context, ranges, metrics, scan_pool).await?;
1313        let n = source.candidates.len() as u128;
1314        let start = (n * partition as u128 / partitions as u128) as usize;
1315        let end = (n * (partition + 1) as u128 / partitions as u128) as usize;
1316        let mut stream = source.read_batches(start..end)?;
1317        while let Some(batch) = stream.try_next().await? {
1318            yield ScanBatch::RecordBatch(batch);
1319        }
1320    })
1321}
